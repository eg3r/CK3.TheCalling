tc_call_set_age_20 = {
	scope = character
	effect = {
		set_variable = { name = tc_call_max_age value = 20 }
	}
}

tc_call_is_age_20 = {
	scope = character
	is_shown = {
		AND = {
			has_variable = tc_call_max_age
			var:tc_call_max_age = 20
		}
	}
}
