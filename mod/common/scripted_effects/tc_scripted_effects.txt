tc_call_lock_options_effect = {
	set_variable = { name = tc_call_lock_age value = 25 }
	if = {
		limit = { has_variable = tc_call_max_age }
		set_variable = { name = tc_call_lock_age value = var:tc_call_max_age }
	}
	set_variable = { name = tc_call_lock_birth value = flag:both }
	if = {
		limit = { has_variable = tc_call_birth }
		set_variable = { name = tc_call_lock_birth value = var:tc_call_birth }
	}
	set_variable = { name = tc_call_lock_faith value = flag:any }
	if = {
		limit = { has_variable = tc_call_faith }
		set_variable = { name = tc_call_lock_faith value = var:tc_call_faith }
	}
	set_variable = { name = tc_call_lock_culture value = flag:any }
	if = {
		limit = { has_variable = tc_call_culture }
		set_variable = { name = tc_call_lock_culture value = var:tc_call_culture }
	}
}

tc_call_clear_lock_effect = {
	remove_variable = tc_call_lock_age
	remove_variable = tc_call_lock_birth
	remove_variable = tc_call_lock_faith
	remove_variable = tc_call_lock_culture
}

tc_settle_called_candidate_effect = {
	if = {
		limit = { is_courtier_of = root }
		add_character_flag = {
			flag = blocked_from_leaving
			years = 10
		}
	}
	else_if = {
		limit = {
			has_character_flag = tc_created_maiden
			is_pool_character = yes
		}
		silent_disappearance_effect = yes
	}
}

tc_find_real_maidens_effect = {
	if = {
		limit = { NOT = { has_game_rule = tc_maiden_source_created } }
		capital_province = { save_scope_as = tc_call_center }
		set_local_variable = { name = tc_found value = 0 }
		every_ruler = {
			limit = {
				tc_in_call_reach_trigger = yes
				in_diplomatic_range = root
			}
			every_courtier_or_guest = {
				limit = { tc_maiden_candidate_trigger = yes }
				add_to_list = tc_maiden_candidates
				change_local_variable = { name = tc_found add = 1 }
			}
		}
		every_pool_character = {
			province = scope:tc_call_center
			limit = {
				NOT = { is_in_list = tc_maiden_candidates }
				tc_maiden_candidate_trigger = yes
			}
			add_to_list = tc_maiden_candidates
			change_local_variable = { name = tc_found add = 1 }
		}
		if = {
			limit = {
				has_local_variable = tc_found
				local_var:tc_found >= 1
			}
			ordered_in_list = {
				list = tc_maiden_candidates
				order_by = tc_maiden_candidate_score_value
				position = 0
				save_scope_as = maiden_1
			}
		}
		if = {
			limit = {
				has_local_variable = tc_found
				local_var:tc_found >= 2
			}
			ordered_in_list = {
				list = tc_maiden_candidates
				order_by = tc_maiden_candidate_score_value
				position = 1
				save_scope_as = maiden_2
			}
		}
		if = {
			limit = {
				has_local_variable = tc_found
				local_var:tc_found >= 3
				NAND = {
					has_game_rule = tc_maiden_congenital_traits_half_chance
					scope:maiden_1 = { tc_has_positive_congenital_trigger = no }
					scope:maiden_2 = { tc_has_positive_congenital_trigger = no }
				}
			}
			ordered_in_list = {
				list = tc_maiden_candidates
				order_by = tc_maiden_candidate_score_value
				position = 2
				save_scope_as = maiden_3
			}
		}
	}
}

tc_fill_maiden_slot_effect = {
	if = {
		limit = { NOT = { exists = scope:$MAIDEN$ } }
		if = {
			limit = { var:tc_call_lock_birth ?= flag:lowborn }
			tc_create_maiden_effect = { MAIDEN = $MAIDEN$ DYNASTY = none }
		}
		else = {
			tc_create_maiden_effect = { MAIDEN = $MAIDEN$ DYNASTY = generate }
		}
	}
}

tc_save_maiden_origin_effect = {
	capital_county = { save_scope_as = tc_maiden_origin }
	random_sub_realm_county = {
		limit = { tc_maiden_origin_county_trigger = yes }
		save_scope_as = tc_maiden_origin
	}
	scope:tc_maiden_origin = {
		culture = { save_scope_as = tc_maiden_culture }
		faith = { save_scope_as = tc_maiden_faith }
	}
	if = {
		limit = { var:tc_call_lock_culture ?= flag:mine }
		culture = { save_scope_as = tc_maiden_culture }
	}
	if = {
		limit = { var:tc_call_lock_faith ?= flag:mine }
		faith = { save_scope_as = tc_maiden_faith }
	}
	else_if = {
		limit = {
			var:tc_call_lock_faith ?= flag:accepted
			root.faith = {
				faith_hostility_level = {
					target = scope:tc_maiden_faith
					value > faith_astray_level
				}
			}
		}
		faith = { save_scope_as = tc_maiden_faith }
	}
}

tc_create_maiden_effect = {
	tc_save_maiden_origin_effect = yes
	create_character = {
		gender = female
		age = { 16 tc_maiden_max_age_value }
		location = root.capital_province
		dynasty = $DYNASTY$
		culture = scope:tc_maiden_culture
		faith = scope:tc_maiden_faith
		random_traits = yes
		diplomacy = {
			min_template_decent_skill
			max_template_decent_skill
		}
		save_scope_as = $MAIDEN$
		after_creation = {
			add_character_flag = {
				flag = tc_created_maiden
				years = 1
			}
			tc_maiden_congenital_traits_effect = yes
			add_character_flag = {
				flag = no_headgear
				years = 1
			}
		}
	}
}

tc_take_maiden_effect = {
	if = {
		limit = { scope:$MAIDEN$ = { has_character_flag = tc_created_maiden } }
		add_courtier = scope:$MAIDEN$
	}
	else = {
		run_interaction = {
			interaction = invite_to_court_interaction
			actor = root
			recipient = scope:$MAIDEN$
			execute_threshold = decline
		}
	}
}

tc_maiden_add_good_trait_effect = {
	if = {
		limit = { num_of_good_genetic_traits = 0 }
		random_list = {
			3000 = { add_trait = beauty_good_1 }
			1000 = { add_trait = beauty_good_2 }
			500 = { add_trait = beauty_good_3 }
			2000 = { add_trait = intellect_good_1 }
			1000 = { add_trait = intellect_good_2 }
			$GENIUS$ = { add_trait = intellect_good_3 }
			2 = { add_trait = pure_blooded }
			1500 = { add_trait = physique_good_1 }
			500 = { add_trait = physique_good_2 }
			500 = { add_trait = fecund }
		}
	}
}
